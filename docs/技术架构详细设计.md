```python
class VectorRetrieval:
    """向量检索"""
    
    def __init__(self, chroma_client):
        self.chroma = chroma_client
        self.embedding_model = "text-embedding-3-small"
    
    async def search(
        self,
        query: str,
        collection_name: str,
        top_k: int = 20,
        filters: Dict = None
    ) -> List[Dict]:
        """向量检索"""
        # 1. 生成查询向量
        query_embedding = await self._embed(query)
        
        # 2. 向量检索
        results = self.chroma.get_collection(collection_name).query(
            query_embeddings=[query_embedding],
            n_results=top_k,
            where=filters
        )
        
        # 3. 格式化结果
        return [
            {
                "id": results["ids"][0][i],
                "content": results["documents"][0][i],
                "metadata": results["metadatas"][0][i],
                "score": results["distances"][0][i]
            }
            for i in range(len(results["ids"][0]))
        ]
    
    async def _embed(self, text: str) -> List[float]:
        """生成向量"""
        from openai import OpenAI
        client = OpenAI()
        response = client.embeddings.create(
            model=self.embedding_model,
            input=text
        )
        return response.data[0].embedding
```

```python
from typing import Dict, List
from app.utils.llm_client import LLMClient

class DocumentGrader:
    """文档相关性评分器"""
    
    def __init__(self, llm_client: LLMClient):
        self.llm = llm_client
        self.threshold = 0.6  # 相关性阈值
    
    async def grade_document(
        self,
        query: str,
        document: str
    ) -> float:
        """对单个文档进行相关性打分（0-1分）"""
        
        grading_prompt = f"""你是一个文档相关性评估专家。请评估以下文档与用户问题的相关性。

用户问题：{query}

检索文档：{document[:800]}

评分标准：
- 1.0: 高度相关，能直接完整回答问题
- 0.7-0.9: 相关性强，包含大部分答案
- 0.5-0.6: 中等相关，包含部分有用信息
- 0.3-0.4: 弱相关，信息不够充分
- 0.0-0.2: 不相关或无关

请仅返回0-1之间的数字分数，例如：0.85
"""
        
        try:
            score_str = await self.llm.complete(
                prompt=grading_prompt,
                temperature=0.1,
                max_tokens=10
            )
            score = float(score_str.strip())
            score = max(0.0, min(1.0, score))  # 确保在0-1之间
            
            logger.info(f"Document relevance score: {score:.2f}")
            return score
            
        except Exception as e:
            logger.error(f"Document grading failed: {e}")
            return 0.5  # 失败时返回中等分数
    
    async def grade_documents(
        self,
        query: str,
        documents: List[Dict]
    ) -> List[Dict]:
        """对多个文档进行评分并排序"""
        
        if not documents:
            return []
        
        # 只对Top文档评分（节省成本）
        top_doc = documents[0]
        score = await self.grade_document(query, top_doc["content"])
        top_doc["relevance_score"] = score
        
        return [top_doc]
    
    def is_relevant(self, score: float) -> bool:
        """判断文档是否足够相关"""
        return score >= self.threshold
```

```python
class WebSearchEngine:
    """联网检索引擎"""

        self.client = TavilyClient(api_key=settings.TAVILY_API_KEY)
        self.max_results = 5

        """判断是否需要联网检索"""

        if relevance_score < 0.6:
            logger.info("Low relevance score, triggering web search")
            return True
        
        # 条件2：包含实时性关键词
        realtime_keywords = [
            "今天", "明天", "最近", "现在", "当前",
            "天气", "气温", "降雨",
            "新闻", "热点", "最新",
            "股价", "汇率", "行情"
        ]
        
        if any(keyword in query for keyword in realtime_keywords):
            logger.info("Realtime keyword detected, triggering web search")
            return True
        
        return False
    
    async def search(self, query: str) -> List[Dict]:
        """执行联网检索"""
        
        try:
            logger.info(f"Searching web for: {query}")
            
            # 调用Tavily搜索API
            results = self.client.search(
                query=query,
                max_results=self.max_results,
                search_depth="basic"
            )
            
            # 提取搜索结果
            web_docs = []
            for result in results.get("results", []):
                web_docs.append({
                    "id": f"web_{hash(result['url'])}",
                    "content": result.get("content", ""),
                    "title": result.get("title", ""),
                    "url": result.get("url", ""),
                    "score": result.get("score", 0.5),
                    "source": "web_search",
                    "relevance_score": 0.7  # 假设联网结果中等相关
                })
            
            logger.info(f"Found {len(web_docs)} web results")
            return web_docs
            
        except Exception as e:
```

from fastapi.responses import StreamingResponse
import json

async def generate_answer_streaming(
    query: str,
    retrieved_docs: List[Dict],
    web_docs: List[Dict] = None
) -> StreamingResponse:
    """生成流式答案（SSE格式）"""
    
    async def event_generator():
        try:
            # 构建上下文
            context = ""
            if retrieved_docs:
                context += "知识库内容：\n" + retrieved_docs[0]["content"][:500]
            if web_docs:
                context += "\n\n联网搜索结果：\n" + "\n".join(
                    [f"[{doc['title']}]({doc['url']})" for doc in web_docs[:3]]
                )
            
            # 构建Prompt
            prompt = f"""基于以下信息回答用户问题：

{context}

用户问题：{query}

请简洁准确地回答："""
            
            # 流式调用LLM
            stream = await llm_client.stream(prompt)
            
            async for chunk in stream:
                # SSE格式输出
                data = {
                    "type": "token",
                    "content": chunk
                }
                yield f"data: {json.dumps(data, ensure_ascii=False)}\n\n"
            
            # 结束标记
            yield f"data: {json.dumps({'type': 'done'}, ensure_ascii=False)}\n\n"
            
        except Exception as e:
            error_data = {"type": "error", "message": str(e)}
            yield f"data: {json.dumps(error_data, ensure_ascii=False)}\n\n"
    
    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream"
    )

```python
class EntityExtractor:
    """实体提取"""
    
    def __init__(self, llm_client):
        self.llm = llm_client
    
    async def extract(self, query: str) -> Dict[str, Any]:
        """提取实体"""
        # 1. 规则提取（快速）
        rule_entities = self._rule_based_extract(query)
        
        # 2. LLM提取（准确）
        llm_entities = await self._llm_based_extract(query)
        
        # 3. 合并结果
        entities = {**rule_entities, **llm_entities}
        
        return entities
    
    def _rule_based_extract(self, query: str) -> Dict:
        """基于规则的实体提取"""
        import re
        entities = {}
        
        # 提取手机号
        phone_pattern = r'1[3-9]\d{9}'
        phones = re.findall(phone_pattern, query)
        if phones:
            entities["phone"] = phones[0]
        
        # 提取邮箱
        email_pattern = r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b'
        emails = re.findall(email_pattern, query)
        if emails:
            entities["email"] = emails[0]
        
        # 提取日期
        date_pattern = r'\d{4}[-/年]\d{1,2}[-/月]\d{1,2}[日]?'
        dates = re.findall(date_pattern, query)
        if dates:
            entities["date"] = dates[0]
        
        return entities
    
    async def _llm_based_extract(self, query: str) -> Dict:
        """基于LLM的实体提取"""
        prompt = f"""从以下文本中提取实体，返回JSON格式。

文本：{query}

需要提取的实体类型：
- product_name: 产品名称
- order_id: 订单号
- user_name: 用户名
- location: 地点

返回格式：
{{
    "product_name": "xxx",
    "order_id": "xxx",
    ...
}}

如果某个实体不存在，则不返回该字段。
"""
        
        response = await self.llm.complete(prompt, temperature=0.1)
        entities = json.loads(response)
        
        return entities
```

```python
class DialogueEngine:
    """对话引擎"""
    
    def __init__(self, llm_client):
        self.llm = llm_client
    
    async def generate(
        self,
        query: str,
        retrieved_docs: List[Dict],
        context: Dict
    ) -> Dict:
        """生成答案"""
        # 1. 构建上下文
        knowledge_context = self._build_knowledge_context(retrieved_docs)
        conversation_history = context.get("conversation_history", "")
        
        # 2. 构建Prompt
        prompt = self._build_prompt(query, knowledge_context, conversation_history)
        
        # 3. LLM生成
        response = await self.llm.complete(
            prompt=prompt,
            temperature=0.7,
            max_tokens=2000,
            stream=True
        )
        
        # 4. 流式返回
        full_answer = ""
        async for chunk in response:
            full_answer += chunk
            yield chunk
        
        # 5. 评估置信度
        confidence = self._estimate_confidence(
            query, full_answer, retrieved_docs
        )
        
        return {
            "content": full_answer,
            "confidence": confidence,
            "sources": [doc["id"] for doc in retrieved_docs[:3]]
        }
    
    def _build_knowledge_context(self, docs: List[Dict]) -> str:
        """构建知识上下文"""
        if not docs:
            return "暂无相关知识库信息"
        
        context = "参考知识库：\n\n"
        for i, doc in enumerate(docs[:3], 1):
            context += f"[{i}] {doc['content']}\n\n"
        
        return context
    
    def _build_prompt(
        self,
        query: str,
        knowledge: str,
        history: str
    ) -> str:
        """构建生成Prompt"""
        prompt = f"""你是一个专业的数字员工助手。请基于知识库回答用户问题。

{knowledge}

对话历史：
{history if history else '无'}

用户问题：{query}

回答要求：
1. 优先使用知识库中的信息
2. 如果知识库中没有相关信息，诚实告知用户
3. 回答简洁清晰，条理分明
4. 使用友好专业的语气

请回答："""
        return prompt
    
    def _estimate_confidence(
        self,
        query: str,
        answer: str,
        docs: List[Dict]
    ) -> float:
        """估算置信度"""
        # 简单的置信度估算逻辑
        if not docs:
            return 0.3  # 无知识库支持，低置信度
        
        if docs[0]["rrf_score"] > 0.5:
            return 0.9  # 高相关度文档
        elif docs[0]["rrf_score"] > 0.3:
            return 0.75  # 中等相关度
        else:
            return 0.6  # 低相关度
```

### 5.1 缓存策略

#### 5.1.1 Redis缓存层

```python
class CacheService:
    """缓存服务"""
    
    def __init__(self, redis_client):
        self.redis = redis_client
        self.default_ttl = 3600  # 1小时
    
    async def get_or_set(
        self,
        key: str,
        fetch_func,
        ttl: int = None
    ):
        """获取缓存或设置"""
        # 尝试从缓存获取
        cached = await self.redis.get(key)
        if cached:
            return json.loads(cached)
        
        # 缓存未命中，执行函数
        value = await fetch_func()
        
        # 设置缓存
        await self.redis.setex(
            key,
            ttl or self.default_ttl,
            json.dumps(value, ensure_ascii=False)
        )
        
        return value
    
    async def invalidate(self, pattern: str):
        """失效缓存"""
        keys = await self.redis.keys(pattern)
        if keys:
            await self.redis.delete(*keys)
```

#### 5.1.2 缓存应用场景

- **FAQ检索缓存**：相同问题缓存答案（TTL: 1小时）
- **意图识别缓存**：相同查询缓存意图（TTL: 30分钟）
- **向量嵌入缓存**：相同文本缓存向量（TTL: 24小时）
- **会话上下文缓存**：活跃会话缓存（TTL: 30分钟）

```python
async def batch_embed_documents(
    documents: List[str],
    batch_size: int = 100
):
    """批量向量化"""
    embeddings = []
    
    for i in range(0, len(documents), batch_size):
        batch = documents[i:i+batch_size]
        
        # 批量调用Embedding API
        response = await openai_client.embeddings.create(
            model="text-embedding-3-small",
            input=batch
        )
        
        batch_embeddings = [item.embedding for item in response.data]
        embeddings.extend(batch_embeddings)
    
    return embeddings
```

### 7.1 API安全

```python
from fastapi import Security, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
import jwt

security = HTTPBearer()

async def verify_token(
    credentials: HTTPAuthorizationCredentials = Security(security)
):
    """验证JWT Token"""
    try:
        payload = jwt.decode(
            credentials.credentials,
            settings.SECRET_KEY,
            algorithms=[settings.JWT_ALGORITHM]
        )
        return payload
    except jwt.ExpiredSignatureError:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Token has expired"
        )
    except jwt.JWTError:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token"
        )
```
