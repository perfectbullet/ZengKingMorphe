## 二、技术架构建议（基于LangGraph）

### 核心工作流（状态机设计）

```python
from langgraph.graph import StateGraph, END
from typing import TypedDict, Annotated, Literal
from operator import add

# 对话状态定义
class AgentState(TypedDict):
    messages: Annotated[list, add]          # 对话历史
    user_query: str                          # 当前问题
    intent: dict                             # 意图识别结果
    entities: dict                           # 实体提取结果
    retrieved_docs: list                     # 召回文档
    search_strategy: Literal["rag", "faq"]  # 检索策略
    confidence: float                        # 置信度
    final_answer: str                        # 最终答案
    need_human: bool                         # 是否转人工

# 构建工作流图
def build_agent_graph():
    graph = StateGraph(AgentState)
    
    # 节点定义
    graph.add_node("intent_recognition", recognize_intent)       # 意图识别
    graph.add_node("entity_extraction", extract_entities)        # 实体提取
    graph.add_node("route_strategy", route_search_strategy)      # 路由决策
    graph.add_node("rag_search", search_rag_docs)               # RAG检索
    graph.add_node("faq_search", search_faq)                    # FAQ检索
    graph.add_node("rerank", rerank_results)                    # 结果重排序
    graph.add_node("answer_generation", generate_answer)         # 答案生成
    graph.add_node("quality_check", check_answer_quality)        # 质量检查
    graph.add_node("sensitive_filter", filter_sensitive_words)   # 敏感词过滤
    
    # 条件路由
    def should_search_rag(state):
        return state["search_strategy"] == "rag"
    
    def need_human_intervention(state):
        return state["confidence"] < 0.6  # 置信度阈值
    
    # 工作流编排
    graph.set_entry_point("intent_recognition")
    graph.add_edge("intent_recognition", "entity_extraction")
    graph.add_edge("entity_extraction", "route_strategy")
    
    # 条件分支
    graph.add_conditional_edges(
        "route_strategy",
        should_search_rag,
        {
            True: "rag_search",
            False: "faq_search"
        }
    )
    
    graph.add_edge("rag_search", "rerank")
    graph.add_edge("faq_search", "rerank")
    graph.add_edge("rerank", "answer_generation")
    graph.add_edge("answer_generation", "quality_check")
    
    # 质量检查后决策
    graph.add_conditional_edges(
        "quality_check",
        need_human_intervention,
        {
            True: END,  # 转人工
            False: "sensitive_filter"
        }
    )
    
    graph.add_edge("sensitive_filter", END)
    
    return graph.compile()
```
